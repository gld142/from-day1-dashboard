# Vue label — plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Un label connecté arrive sur son roster (bloc « Le label ce mois-ci »), garde la skin structure, change d'artiste depuis la barre du haut, et lit Revenus / Streams en « Tout le label ».

**Architecture:** La vue (`artist` | `label`) est calculée à la connexion depuis `workspaces.kind` et le rôle, portée par `CustomerInfo`. Les relevés des artistes sont fusionnés côté serveur en un relevé virtuel (`UserData` avec un champ `scope`) que les pages Revenus / Streams existantes affichent quand l'URL porte `?perimetre=label`. Le roster calcule le dernier mois commun dans `summarizeRoster`. Aucune migration.

**Tech Stack:** Next.js (app router, Server Components + actions), React 19, Supabase (RLS, `userDb` / `adminDb`), vitest, Tailwind, recharts, lucide-react. Spec : `docs/superpowers/specs/2026-10-22-vue-label-design.md`.

**Règles du dépôt à respecter (AGENTS.md + mémoire) :** `git add -- <fichiers>` explicite, jamais `-A`. Ne jamais lancer `vercel --prod` depuis `from-day1-dashboard`. Base de test remise à zéro à la fin. Lire `node_modules/next/dist/docs/` avant d'utiliser une API Next inhabituelle.

/**
 * La vue du dashboard d'un client : « artist » (un artiste, son relevé) ou
 * « label » (une structure : roster en accueil, totaux, menu d'artistes).
 *
 * Décidé avec Gaël le 22/10/2026 : c'est le TYPE DÉCLARÉ À L'ACHAT qui tranche
 * (`workspaces.kind`), pas le nombre d'artistes. Un artiste invité (rôle
 * « artist ») reste toujours en vue artiste.
 */
export const WORKSPACE_KINDS = ["artist", "label", "management", "publisher", "distributor", "other"] as const;
export type WorkspaceKind = (typeof WORKSPACE_KINDS)[number];
export type DashboardView = "artist" | "label";
export type MemberRole = "owner" | "member" | "artist";

export const KIND_LABEL: Record<WorkspaceKind, string> = {
  artist: "Artiste",
  label: "Label",
  management: "Management",
  publisher: "Éditeur",
  distributor: "Distributeur",
  other: "Autre",
};

export function isWorkspaceKind(v: string): v is WorkspaceKind {
  return (WORKSPACE_KINDS as readonly string[]).includes(v);
}

export function viewFor(kind: WorkspaceKind, role: MemberRole): DashboardView {
  if (role === "artist") return "artist";
  return kind === "artist" ? "artist" : "label";
}
```

/** Accueil du mode client : le relevé mensuel pour un artiste, le roster pour une structure. */
export const CUSTOMER_HOME = "/mon-releve";
export const LABEL_HOME = "/roster";
export function customerHome(view: DashboardView): string {
  return view === "label" ? LABEL_HOME : CUSTOMER_HOME;
}

/**
 * Où revenir après avoir rouvert un autre artiste (`/espace/ouvrir/[id]?vers=`) :
 * seulement une page du dashboard ouverte à ce rôle, sinon l'accueil de la vue.
 * `perimetre` est retiré : on vient d'ouvrir un artiste, le périmètre redevient lui.
 */
export function returnPath(vers: string | null, role: CustomerInfo["role"], view: DashboardView): string {
  const home = customerHome(view);
  if (!vers) return home;
  if (vers === "import") return "/import";
  if (!vers.startsWith("/") || vers.startsWith("//") || vers.includes("..") || vers.includes("\\")) return home;
  const [path, query = ""] = vers.split("?", 2);
  if (!isCustomerPath(path, role)) return home;
  const params = new URLSearchParams(query);
  params.delete("perimetre");
  const q = params.toString();
  return q ? `${path}?${q}` : path;
}
```

/**

 * session (RLS) — puis revient sur la page demandée (`vers`), si elle est
 * ouverte à ce rôle. Le cookie est posé AVANT de relire la vue : `customerView`

export async function GET(request: NextRequest, ctx: { params: Promise<{ artistId: string }> }) {
  const { artistId } = await ctx.params;
  const db = await userDb();
  const { data } = await db.from("artists").select("id, workspace_id").eq("id", artistId).maybeSingle();

}

function latestMonth(artists: RosterArtist[], end: string | null): LatestMonth | null {
  if (!end) return null;
  const prev = addMonths(end, -1);
  const covered: LatestMonth["covered"] = [];
  const uncovered: LatestMonth["uncovered"] = [];
  let revenue = 0;
  let streams = 0;
  let prevRevenue = 0;
  let prevStreams = 0;
  let currentOnBoth = 0;
  let onArtists = 0;
  for (const a of artists) {
    const s = a.statement;
    if (s && s.months.includes(end)) {
      covered.push({ id: a.id, name: a.name });
      revenue += revenueOf(s, end);
      streams += streamsOf(s, end);
      if (s.months.includes(prev)) {
        onArtists++;
        prevRevenue += revenueOf(s, prev);
        prevStreams += streamsOf(s, prev);
        currentOnBoth += revenueOf(s, end);
      }
    } else {
      uncovered.push({ id: a.id, name: a.name, reason: s && s.months.length ? "behind" : "missing" });
    }
  }
  return {
    month: end,
    revenue,
    streams,
    per1000: streams > 0 ? (revenue / streams) * 1000 : null,
    covered,
    uncovered,
    previous: onArtists
      ? { revenue: prevRevenue, streams: prevStreams, change: prevRevenue > 0 ? currentOnBoth / prevRevenue - 1 : null, onArtists }
      : null,
  };
}

/**
 * Relevé virtuel d'une structure : les derniers relevés de ses artistes,
 * additionnés en un seul `UserData` que les pages Revenus / Streams affichent
 * tel quel (vue label, `?perimetre=label`).
 *
 * Règles (spec du 22/10/2026) :
 *  - mois = union ; sommes partout où le champ existe chez tous ;
 *  - titres renommés « Titre · Artiste » (deux « Intro » restent distincts) ;
 *  - un détail optionnel absent chez UN artiste est omis, et l'artiste nommé
 *    dans `missingDetail` — on n'affiche pas un « total » qui n'en est pas un ;
 *  - tout est déjà en euros à l'import : devise EUR, taux 1, pas de `fx`.
 */
import type { UserData } from "./store";

export type MergeEntry = { id: string; name: string; data: UserData };
export type MissingDetail = "platforms" | "titles" | "countries" | "detail";
export type LabelStatement = {
  /** Le relevé virtuel ; null sans aucun relevé. */
  data: UserData | null;
  artists: number;
  withStatement: number;
  /** mois → nombre d'artistes dont le relevé couvre ce mois */
  coverage: Record<string, number>;
  /** Période couverte par chaque relevé (pour expliquer les mois partiels). */
  spans: { id: string; name: string; from: string; to: string }[];
  missingDetail: { id: string; name: string; what: MissingDetail[] }[];
};

const cents = (n: number) => Math.round(n * 100) / 100;

function addInto(target: Record<string, number>, source: Record<string, number | undefined> | undefined) {
  for (const [k, v] of Object.entries(source ?? {})) if (v !== undefined) target[k] = cents((target[k] ?? 0) + v);
}
function addNested(
  target: Record<string, Record<string, number>>,
  source: Record<string, Record<string, number | undefined>> | undefined,
  key: (k: string) => string = (k) => k,
) {
  for (const [k, inner] of Object.entries(source ?? {})) addInto((target[key(k)] ??= {}), inner);
}

const DETAIL_FIELDS: Record<MissingDetail, (keyof UserData)[]> = {
  platforms: ["revenueByMonthDsp"],
  titles: ["trackRevenue"],
  countries: ["countryRevenue"],
  detail: ["trackMonthStreams"],
};

export function mergeStatements(entries: MergeEntry[], opts: { artistName: string; artists: number }): LabelStatement {
  const coverage: Record<string, number> = {};
  const spans: LabelStatement["spans"] = [];
  for (const e of entries) {
    const months = [...e.data.months].sort();
    for (const m of months) coverage[m] = (coverage[m] ?? 0) + 1;
    if (months.length) spans.push({ id: e.id, name: e.name, from: months[0], to: months[months.length - 1] });
  }
  const missingDetail: LabelStatement["missingDetail"] = entries
    .map((e) => ({ id: e.id, name: e.name, what: (Object.keys(DETAIL_FIELDS) as MissingDetail[]).filter((w) => DETAIL_FIELDS[w].some((f) => e.data[f] === undefined)) }))
    .filter((m) => m.what.length > 0);
  if (!entries.length) return { data: null, artists: opts.artists, withStatement: 0, coverage, spans, missingDetail };

  const all = (what: MissingDetail) => !missingDetail.some((m) => m.what.includes(what));
  const titled = (e: MergeEntry) => (t: string) => `${t} · ${e.name}`;

  const streamsByMonth: Record<string, Record<string, number>> = {};
  const revenueByMonth: Record<string, number> = {};
  const trackStreams: Record<string, number> = {};
  const countryStreams: Record<string, number> = {};
  const revenueByMonthDsp: Record<string, Record<string, number>> = {};
  const trackRevenue: Record<string, number> = {};
  const countryRevenue: Record<string, number> = {};
  const trackMonthStreams: Record<string, Record<string, number>> = {};
  const trackMonthRevenue: Record<string, Record<string, number>> = {};
  const countryMonthStreams: Record<string, Record<string, number>> = {};
  const countryMonthRevenue: Record<string, Record<string, number>> = {};
  const trackIsrc: Record<string, string> = {};
  const saleReportRevenue: Record<string, number> = {};
  const trackSpotify12: Record<string, number> = {};
  let withheld: number | undefined;
  let importedAt = "";
  let totalStreams = 0;
  let totalRevenueEur = 0;

  for (const e of entries) {
    const d = e.data;
    const key = titled(e);
    addNested(streamsByMonth, d.streamsByMonth as Record<string, Record<string, number | undefined>>);
    addInto(revenueByMonth, d.revenueByMonth);
    addInto(trackStreams, Object.fromEntries(Object.entries(d.trackStreams).map(([t, v]) => [key(t), v])));
    addInto(countryStreams, d.countryStreams);
    if (all("platforms")) addNested(revenueByMonthDsp, d.revenueByMonthDsp as Record<string, Record<string, number | undefined>>);
    if (all("titles")) addInto(trackRevenue, Object.fromEntries(Object.entries(d.trackRevenue ?? {}).map(([t, v]) => [key(t), v])));
    if (all("countries")) addInto(countryRevenue, d.countryRevenue);
    if (all("detail")) {
      addNested(trackMonthStreams, d.trackMonthStreams, key);
      addNested(trackMonthRevenue, d.trackMonthRevenue, key);
      addNested(countryMonthStreams, d.countryMonthStreams);
      addNested(countryMonthRevenue, d.countryMonthRevenue);
      for (const [t, isrc] of Object.entries(d.trackIsrc ?? {})) trackIsrc[key(t)] = isrc;
      addInto(saleReportRevenue, d.saleReportRevenue);
    }
    // Indépendant du détail mois par mois : gardé si la fenêtre est la même chez tous (voir sameWindow).
    for (const [t, v] of Object.entries(d.trackSpotify12 ?? {})) trackSpotify12[key(t)] = v;
    if (d.withheldEur !== undefined) withheld = cents((withheld ?? 0) + d.withheldEur);
    if (d.importedAt > importedAt) importedAt = d.importedAt;
    totalStreams += d.totalStreams;
    totalRevenueEur += d.totalRevenueEur;
  }

  const windows = entries.map((e) => JSON.stringify(e.data.spotifyWindow ?? null));
  const sameWindow = windows[0] !== "null" && windows.every((w) => w === windows[0]);

  const data: UserData = {
    version: 1,
    artistName: opts.artistName,
    importedAt,
    format: "generic",
    eurRate: 1,
    sourceCurrency: "EUR",
    months: Object.keys(coverage).sort(),
    streamsByMonth: streamsByMonth as UserData["streamsByMonth"],
    revenueByMonth,
    trackStreams,
    countryStreams,
    ...(all("platforms") ? { revenueByMonthDsp: revenueByMonthDsp as UserData["revenueByMonthDsp"] } : {}),
    ...(all("titles") ? { trackRevenue } : {}),
    ...(all("countries") ? { countryRevenue } : {}),
    ...(all("detail") ? { trackMonthStreams, trackMonthRevenue, countryMonthStreams, countryMonthRevenue, trackIsrc, saleReportRevenue } : {}),
    ...(sameWindow ? { trackSpotify12, spotifyWindow: entries[0].data.spotifyWindow } : {}),
    ...(withheld !== undefined ? { withheldEur: withheld } : {}),
    totalStreams,
    totalRevenueEur: cents(totalRevenueEur),
    active: true,
    scope: { kind: "label", artists: opts.artists, withStatement: entries.length, coverage },
  };
  return { data, artists: opts.artists, withStatement: entries.length, coverage, spans, missingDetail };
}

export function coverageNote(points: MonthPoint[], label: LabelStatement, month: (m: string) => string): string | null {
  const partial = points.filter((p) => p.partial && !p.missing);
  if (!partial.length) return null;
  const byCount = new Map<number, string[]>();
  for (const p of partial) {
    const c = label.coverage[p.month] ?? 0;
    byCount.set(c, [...(byCount.get(c) ?? []), p.month]);

/** Le type d'espace décide de la vue du dashboard (artiste / label) — propriétaire seulement. */
export async function setWorkspaceKind(_: FormState, form: FormData): Promise<FormState> {
  const workspaceId = String(form.get("workspaceId") ?? "");
  const kind = String(form.get("kind") ?? "");
  if (!isWorkspaceKind(kind)) return { status: "error", message: "Type d'espace inconnu." };
  if (!(await ownerOf(workspaceId))) return { status: "error", message: "Seul le propriétaire de l'espace peut changer son type." };
  const { error } = await adminDb().from("workspaces").update({ kind }).eq("id", workspaceId);
  if (error) {
    console.error("[espace/setWorkspaceKind]", error);
    return { status: "error", message: "Le type n'a pas pu être enregistré." };
  }
  revalidatePath("/espace");
  revalidatePath("/", "layout");
  return { status: "ok" };
}
